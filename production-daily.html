async function fetchAndRenderData() { const dateVal = document.getElementById('dailyDate').value; const lineVal = document.getElementById('filterLine').value; const shiftVal = document.getElementById('filterShift').value; const loading = document.getElementById('loadingStatus'); loading.classList.remove('hidden'); let currentData = []; try { let query = supabaseClient.from('daily_production').select('*'); // หากต้องการลองดึงข้อมูลทั้งหมดมาเช็คก่อน ให้ลองปิดบรรทัด if (dateVal) ชั่วคราว if (dateVal) query = query.eq('date', dateVal); if (lineVal !== 'ALL') query = query.eq('line', lineVal); if (shiftVal !== 'ALL') query = query.ilike('shift', `%${shiftVal}%`); const { data, error } = await query.order('time_slot', { ascending: true }); if (error) { console.error('❌ Supabase Query Error:', error.message); } else { console.log('✅ ข้อมูลที่ดึงได้จาก Supabase:', data); currentData = data || []; } } catch (err) { console.error('❌ Exception Error:', err.message); } finally { updateKPIs(currentData); renderHourlyChart(currentData); renderLinePerformance(currentData); renderHourlyTable(currentData); loading.classList.add('hidden'); } }
